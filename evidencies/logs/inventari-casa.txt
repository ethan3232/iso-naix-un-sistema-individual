$ python3 -c '
import time
n=512*1024*1024; a=bytearray(b"\x55")*n
t=time.time(); b=bytes(a); d=time.time()-t
print(f"Copia de 512 MiB: {d:.2f} s -> {n/d/1e9:.1f} GB/s")
ok = b.count(b"\x55")==n
print("Patrón 0x55 verificado sin errores" if ok else "ERROR: el patrón no coincide")'
